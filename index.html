<script>
    // Dữ liệu từ vựng mẫu (Bạn có thể tự thêm từ mới vào đây)
    const vocabData = [
        { hanzi: "你好", pinyin: "nǐ hǎo", meaning: "Xin chào" },
        { hanzi: "谢谢", pinyin: "xièxiè", meaning: "Cảm ơn" },
        { hanzi: "再见", pinyin: "zàijiàn", meaning: "Tạm biệt" },
        { hanzi: "爱", pinyin: "ài", meaning: "Yêu, thích" },
        { hanzi: "学习", pinyin: "xuéxí", meaning: "Học tập" },
        { hanzi: "中国", pinyin: "zhōngguó", meaning: "Trung Quốc" }
    ];

    let currentList = vocabData;
    let currentIndex = 0;
    let isFlipped = false;
    let hanziWriterInstance = null;

    // Khởi tạo HanziWriter
    function initWriter(char) {
        const container = document.getElementById('character-target');
        container.innerHTML = ""; // Xóa chữ cũ
        hanziWriterInstance = HanziWriter.create('character-target', char, {
            width: 120,
            height: 120,
            padding: 5,
            showOutline: true,
            strokeAnimationSpeed: 1,
            delayBetweenStrokes: 300,
            strokeColor: '#4f46e5'
        });
    }

    // Cập nhật Flashcard & HanziWriter theo index hiện tại
    function updateCard() {
        isFlipped = false;
        document.getElementById('cardElement').classList.remove('flipped');
        
        const item = currentList[currentIndex];
        document.getElementById('fc-hanzi').innerText = item.hanzi;
        document.getElementById('fc-pinyin').innerText = item.pinyin;
        document.getElementById('fc-meaning').innerText = item.meaning;
        document.getElementById('fc-counter').innerText = `${currentIndex + 1} / ${currentList.length}`;

        // Lấy ký tự đầu tiên của từ để vẽ nét (nếu từ dài hơn 1 chữ Hán)
        initWriter(item.hanzi.charAt(0));
    }

    function flipCard() {
        isFlipped = !isFlipped;
        document.getElementById('cardElement').classList.toggle('flipped', isFlipped);
    }

    function nextCard() {
        currentIndex = (currentIndex + 1) % currentList.length;
        updateCard();
    }

    function prevCard() {
        currentIndex = (currentIndex - 1 + currentList.length) % currentList.length;
        updateCard();
    }

    // Phát âm chuẩn bằng Web Speech API của trình duyệt
    function audioPlay(text) {
        if ('speechSynthesis' in window) {
            const utterance = new SpeechSynthesisUtterance(text);
            utterance.lang = 'zh-CN'; // Thiết lập ngôn ngữ Tiếng Trung
            utterance.rate = 0.8; // Đọc chậm một chút cho dễ nghe
            window.speechSynthesis.speak(utterance);
        } else {
            alert("Trình duyệt của bạn không hỗ trợ phát âm thanh tự động!");
        }
    }
    window.playAudio = audioPlay;

    // Chuyển Tab giao diện
    function switchTab(tabName) {
        document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
        document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));

        if (tabName === 'flashcard') {
            document.querySelector('.tabs button:nth-child(1)').classList.add('active');
            document.getElementById('flashcard-tab').classList.add('active');
            updateCard();
        } else {
            document.querySelector('.tabs button:nth-child(2)').classList.add('active');
            document.getElementById('quiz-tab').classList.add('active');
            initQuiz();
        }
    }

    // Logic phần Trắc nghiệm (Quiz)
    let quizIndex = 0;
    let score = 0;
    let shuffledQuizList = [];

    function initQuiz() {
        quizIndex = 0;
        score = 0;
        // Xáo trộn danh sách từ vựng để làm câu hỏi
        shuffledQuizList = [...vocabData].sort(() => Math.random() - 0.5);
        loadQuizQuestion();
    }

    function loadQuizQuestion() {
        document.getElementById('next-quiz-btn').style.display = 'none';
        document.getElementById('quiz-score-display').innerText = '';

        if (quizIndex >= shuffledQuizList.length) {
            document.getElementById('quiz-question-text').innerText = "🎉 Hoàn thành bài trắc nghiệm!";
            document.getElementById('quiz-options').innerHTML = '';
            document.getElementById('quiz-score-display').innerText = `Điểm số của bạn: ${score} / ${vocabData.length}`;
            document.getElementById('next-quiz-btn').innerText = "Làm lại bài";
            document.getElementById('next-quiz-btn').style.display = 'block';
            document.getElementById('next-quiz-btn').onclick = initQuiz;
            return;
        }

        document.getElementById('next-quiz-btn').innerText = "Câu tiếp theo";
        document.getElementById('next-quiz-btn').onclick = nextQuiz;

        const currentQ = shuffledQuizList[quizIndex];
        document.getElementById('quiz-progress').innerText = `Câu hỏi ${quizIndex + 1} / ${shuffledQuizList.length}`;
        document.getElementById('quiz-question-text').innerHTML = `Nghĩa của từ <span style="color:var(--primary);">${currentQ.hanzi}</span> (${currentQ.pinyin}) là gì?`;

        // Tạo 3 đáp án nhiễu + 1 đáp án đúng
        let options = [currentQ.meaning];
        while (options.length < 4 && options.length < vocabData.length) {
            let randomOpt = vocabData[Math.floor(Math.random() * vocabData.length)].meaning;
            if (!options.includes(randomOpt)) {
                options.push(randomOpt);
            }
        }
        // Xáo trộn vị trí đáp án
        options.sort(() => Math.random() - 0.5);

        const optionsContainer = document.getElementById('quiz-options');
        optionsContainer.innerHTML = '';
        options.forEach(opt => {
            const btn = document.createElement('button');
            btn.className = 'option-btn';
            btn.innerText = opt;
            btn.onclick = () => checkAnswer(btn, opt, currentQ.meaning);
            optionsContainer.appendChild(btn);
        });
    }

    function checkAnswer(selectedBtn, selectedOpt, correctOpt) {
        const allBtns = document.querySelectorAll('.option-btn');
        allBtns.forEach(b => b.disabled = true); // Khóa các nút sau khi chọn

        if (selectedOpt === correctOpt) {
            selectedBtn.classList.add('correct');
            score++;
        } else {
            selectedBtn.classList.add('wrong');
            // Tìm và highlight đáp án đúng
            allBtns.forEach(b => {
                if (b.innerText === correctOpt) b.classList.add('correct');
            });
        }
        document.getElementById('next-quiz-btn').style.display = 'block';
    }

    function nextQuiz() {
        quizIndex++;
        loadQuizQuestion();
    }

    // Chạy mặc định lần đầu
    updateCard();
</script>
